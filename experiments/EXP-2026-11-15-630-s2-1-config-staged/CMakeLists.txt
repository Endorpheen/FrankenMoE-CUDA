# Staged S1.1 build: llama.cpp (CPU only) with one Goose module behind a flag.
#
#   OFF (default): the native json-schema-to-grammar.cpp body is selected, the
#                  Goose object is not compiled and not linked.
#   ON           : json_schema_to_grammar() dispatches to the Goose module
#                  (goose/json/format-grammar.goose -> goose-gen/json-schema-grammar.c)
#                  through common/fg-goose-schema-bridge.cpp. The native body
#                  stays compiled under json_schema_to_grammar_native().
#
# Nothing here touches the production tree: this file, tests/ and goose-gen/
# live only in the staged copy.
cmake_minimum_required(VERSION 3.21)

project(fg-goose-staged LANGUAGES C CXX)

option(FG_GOOSE_JSON_SCHEMA
       "Select the Goose json-schema-to-grammar module instead of the native converter"
       OFF)

# S2.1: the config-consistency gate. OFF (default) runs the native validate()
# body; ON runs cfg_validate() from the composed Goose unit and reproduces the
# native English messages in the carrier. The native body stays compiled in both
# configurations (validate_native), so switching the flag off is a rebuild.
option(FG_GOOSE_CONFIG
       "Select the Goose config validator (cfg_validate) instead of the native validate()"
       OFF)

# Gate optimization level. The repository convention is the gate at -O1 and the
# mutation runs at -O0; this variable is how the two are expressed, so a mutant
# never needs a hand-edited CMakeLists.
set(FG_GATE_OPT "-O1" CACHE STRING "Optimization level of the config gate and the Goose object")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# llama.cpp scope: CPU only, libraries only, no server, no model tools, no
# network, no LLGuidance. These are the same switches the base build uses.
set(LLAMA_BUILD_COMMON   ON  CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS    OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS    OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER   OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP      OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL        OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS     OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE     OFF CACHE BOOL "" FORCE)
set(GGML_CUDA            OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE          ON  CACHE BOOL "" FORCE)

add_subdirectory(llama.cpp)

# --- the one Goose module ---------------------------------------------------

if(FG_GOOSE_JSON_SCHEMA)
    # Generated C is one translation unit and one object of its own; the Goose
    # runtime is self-contained (no host hook is needed by this module).
    add_library(fg-goose-json-schema OBJECT goose-gen/json-schema-grammar.c)
    set_target_properties(fg-goose-json-schema PROPERTIES
        C_STANDARD 11
        C_STANDARD_REQUIRED ON
        POSITION_INDEPENDENT_CODE ON)
    target_include_directories(fg-goose-json-schema PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/goose-gen)
    # Generated code: keep llama.cpp's warning set out of it.
    target_compile_options(fg-goose-json-schema PRIVATE -w)

    target_sources(llama-common PRIVATE common/fg-goose-schema-bridge.cpp)
    target_compile_definitions(llama-common PRIVATE FG_GOOSE_JSON_SCHEMA=1)
    target_include_directories(llama-common PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/goose-gen)
    target_link_libraries(llama-common PRIVATE fg-goose-json-schema)
endif()

# --- the config validator (S2.1) ---------------------------------------------
#
# runtime/core/src/config.cpp is pure policy: it includes only bmoe/config.h, so
# it needs no llama.cpp, no model and no GPU. The gate binary therefore links
# libfg-config-core alone; the staged llama.cpp targets are built by the same
# command but are not on its link line.

add_library(fg-config-core SHARED runtime/core/src/config.cpp)
target_include_directories(fg-config-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/runtime/core/include
                                                ${CMAKE_CURRENT_SOURCE_DIR}/common)
# The repository convention: the gate runs at -O1 (mutations at -O0). Release
# puts -O3 on the command line first; this wins because it comes later.
target_compile_options(fg-config-core PRIVATE ${FG_GATE_OPT})
set_target_properties(fg-config-core PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)

if(FG_GOOSE_CONFIG)
    # The composed Goose unit: arch-registry + config-validate + config-resolve,
    # generated by `goose --standalone` from goose-gen/config-slice.goose.
    add_library(fg-goose-config OBJECT goose-gen/config-slice.c)
    set_target_properties(fg-goose-config PROPERTIES
        C_STANDARD 11
        C_STANDARD_REQUIRED ON
        POSITION_INDEPENDENT_CODE ON)
    target_compile_options(fg-goose-config PRIVATE ${FG_GATE_OPT} -w)
    target_include_directories(fg-config-core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/goose-gen)

    target_sources(fg-config-core PRIVATE common/fg-goose-config-bridge.cpp)
    target_compile_definitions(fg-config-core PRIVATE FG_GOOSE_CONFIG=1)
    target_link_libraries(fg-config-core PRIVATE fg-goose-config)
endif()

# --- gate -------------------------------------------------------------------

add_executable(fg-config-smoke tests/fg_config_smoke.cpp)
target_link_libraries(fg-config-smoke PRIVATE fg-config-core)
target_compile_options(fg-config-smoke PRIVATE ${FG_GATE_OPT})
if(FG_GOOSE_CONFIG)
    target_compile_definitions(fg-config-smoke PRIVATE FG_GOOSE_CONFIG=1)
endif()

add_executable(fg-goose-smoke tests/fg_goose_smoke.cpp)
target_link_libraries(fg-goose-smoke PRIVATE llama-common)
if(FG_GOOSE_JSON_SCHEMA)
    target_compile_definitions(fg-goose-smoke PRIVATE FG_GOOSE_JSON_SCHEMA=1)
endif()

# Prints the selected grammar text per corpus schema; the base and OFF builds
# must print the same text (switch-off parity).
add_executable(fg-goose-dump tests/fg_goose_dump.cpp)
target_link_libraries(fg-goose-dump PRIVATE llama-common)

message(STATUS "FG_GOOSE_JSON_SCHEMA = ${FG_GOOSE_JSON_SCHEMA}")
message(STATUS "FG_GOOSE_CONFIG = ${FG_GOOSE_CONFIG}")
message(STATUS "FG_GATE_OPT = ${FG_GATE_OPT}")
